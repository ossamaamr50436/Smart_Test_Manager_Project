السكربت scripts/delete-non-super-admin-users.ts:

typescript
import { PrismaClient, Role } from "@prisma/client";

const p = new PrismaClient();

async function main() {
  const KEEP_EMAIL = "ossamaamr50436@gmail.com";

  console.log("🗑️  بدء عملية حذف المستخدمين...\n");

  // 1. تحقق من وجود المستخدم الواجب الاحتفاظ به
  const keepUser = await p.user.findUnique({
    where: { email: KEEP_EMAIL },
  });

  if (!keepUser) {
    throw new Error(`❌ لم أجد: ${KEEP_EMAIL}`);
  }

  console.log("✅ سيتم الاحتفاظ بـ:", keepUser.email, "| Role:", keepUser.role);

  // 2. جلب المستخدمين المراد حذفهم
  const usersToDelete = await p.user.findMany({
    where: { email: { not: KEEP_EMAIL } },
    select: { id: true, email: true, role: true },
  });

  if (usersToDelete.length === 0) {
    console.log("ℹ️ لا يوجد مستخدمون للحذف");
    return;
  }

  console.log(`\n📋 سيتم حذف ${usersToDelete.length} مستخدم:`);
  usersToDelete.forEach((u) => console.log(`   - ${u.email} (${u.role})`));

  const idsToDelete = usersToDelete.map((u) => u.id);

  // 3. معالجة Foreign Keys

  // 3.1 AuditLog.userId — nullable، نضبطه على null
  const auditLogs = await p.auditLog.updateMany({
    where: { userId: { in: idsToDelete } },
    data: { userId: null },
  });
  console.log(`\n✅ AuditLogs.userId → null: ${auditLogs.count}`);

  // 3.2 Notification.userId — غير nullable، نحذف الإشعارات
  const notifications = await p.notification.deleteMany({
    where: { userId: { in: idsToDelete } },
  });
  console.log(`✅ Notifications deleted: ${notifications.count}`);

  // 3.3 Assessment.evaluatorId — نحذف التقييمات
  const assessments = await p.assessment.deleteMany({
    where: { evaluatorId: { in: idsToDelete } },
  });
  console.log(`✅ Assessments deleted: ${assessments.count}`);

  // 3.4 ExamSession — نحذف الجلسات (teacher1Id أو teacher2Id)
  const sessionsAsTeacher = await p.examSession.deleteMany({
    where: {
      OR: [
        { teacher1Id: { in: idsToDelete } },
        { teacher2Id: { in: idsToDelete } },
      ],
    },
  });
  console.log(`✅ ExamSessions deleted: ${sessionsAsTeacher.count}`);

  // 3.5 Committee — نحذف اللجان (teacher1Id أو teacher2Id)
  const committeesAsTeacher = await p.committee.deleteMany({
    where: {
      OR: [
        { teacher1Id: { in: idsToDelete } },
        { teacher2Id: { in: idsToDelete } },
      ],
    },
  });
  console.log(`✅ Committees deleted: ${committeesAsTeacher.count}`);

  // 3.6 Certificate.issuedById — nullable، نضبطه على null
  const certsIssued = await p.certificate.updateMany({
    where: { issuedById: { in: idsToDelete } },
    data: { issuedById: null, signedById: null },
  });
  console.log(`✅ Certificates.issuedById → null: ${certsIssued.count}`);

  // 3.7 Institution.users — علاقة، لا نحتاج فعل شيء

  // 4. الآن احذف المستخدمين
  const deleted = await p.user.deleteMany({
    where: { id: { in: idsToDelete } },
  });
  console.log(`\n🗑️  تم حذف ${deleted.count} مستخدم`);

  // 5. التحقق النهائي
  const remaining = await p.user.findMany({
    select: { email: true, role: true, tenantId: true },
  });
  console.log("\n✅ المستخدمون المتبقون:");
  remaining.forEach((u) =>
    console.log(`   - ${u.email} (${u.role}) | tenantId: ${u.tenantId || "null"}`)
  );
}

main()
  .catch((e) => {
    console.error("❌ خطأ:", e);
    process.exitCode = 1;
  })
  .finally(() => p.$disconnect());
