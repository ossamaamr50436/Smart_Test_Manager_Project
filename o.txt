import type { Prisma } from "@prisma/client";

export async function withTenantContext<T>(
  user: SessionUser,
  fn: (tx: Prisma.TransactionClient) => Promise<T>
): Promise<T> {
  return prisma.$transaction(async (tx) => {
    if (user.role !== "SUPER_ADMIN" && user.tenantId) {
      await tx.$executeRaw`SELECT set_config('app.current_tenant_id', ${user.tenantId}, true)`;
      await tx.$executeRaw`SELECT set_config('app.current_user_role', ${user.role}, true)`;
    } else if (user.role === "SUPER_ADMIN") {
      await tx.$executeRaw`SELECT set_config('app.current_user_role', 'SUPER_ADMIN', true)`;
    }
    return fn(tx);
  });
}

export async function checkMultiLevelRateLimit(
  user: SessionUser | null,
  ipAddress: string,
  action: string
): Promise<void> {
  await checkRateLimit(`ip:${ipAddress}:${action}`, 100);
  if (user) await checkRateLimit(`user:${user.id}:${action}`, 50);
  if (user?.tenantId) await checkRateLimit(`tenant:${user.tenantId}:${action}`, 500);
}
