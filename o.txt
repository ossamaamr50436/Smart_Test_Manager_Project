المشكلة المحتملة:

المطلوب:

تأكد من أن دالة handleSubmit تستدعي signIn بشكل صحيح:

tsx
async function handleSubmit(e: React.FormEvent) {
  e.preventDefault();
  setLoading(true);
  setError("");

  const result = await signIn("credentials", {
    email,
    password,
    redirect: false,
  });

  setLoading(false);

  if (result?.error) {
    setError("بيانات الدخول غير صحيحة");
    return;
  }

  router.push("/dashboard");
  router.refresh();
}
إذا كان الكود مختلفاً، أعد كتابته بالكامل كما هو موضح أعلاه.

3. فحص وتصحيح ملف auth.ts
المشكلة المحتملة:

دالة authorize لا تعمل بشكل صحيح.

bcrypt.compare يفشل بسبب تنسيق كلمة المرور.

المطلوب:

اقرأ ملف auth.ts بالكامل.

تأكد من أن دالة authorize تحتوي على:

typescript
async function authorize(credentials) {
  const user = await prisma.user.findUnique({
    where: { email: credentials.email },
  });

  if (!user) return null;

  const isValid = await bcrypt.compare(credentials.password, user.password);
  if (!isValid) return null;

  return {
    id: user.id,
    name: user.name,
    email: user.email,
    role: user.role,
  };
}
تأكد من أن bcryptjs مستوردة بشكل صحيح.

تأكد من أن prisma مستوردة.

4. فحص وتصحيح ملف auth.config.ts
المشكلة المحتملة:

callbacks لا تعمل بشكل صحيح.

المطلوب:

اقرأ ملف auth.config.ts.

تأكد من أن callbacks.jwt و callbacks.session تحتويان على:

typescript
callbacks: {
  async jwt({ token, user }) {
    if (user) {
      token.id = user.id;
      token.role = user.role;
    }
    return token;
  },
  async session({ session, token }) {
    if (session.user) {
      session.user.id = token.id;
      session.user.role = token.role;
    }
    return session;
  },
}
