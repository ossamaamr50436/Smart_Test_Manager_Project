// ========== النموذج الجديد (Input Validation) ==========

export const createExamModelInput = z.object({
  number: z.number().int().min(1).max(100),
  startSurah: z.string().min(1),
  startVerse: z.number().int().min(1),
  startText: z.string().optional(),
  endSurah: z.string().min(1),
  endVerse: z.number().int().min(1),
  endText: z.string().optional(),
  sections: z.array(z.object({
    id: z.string(),
    startVerse: z.string(), // "الفاتحة:1"
    endVerse: z.string(),   // "الفاتحة:3"
    lines: z.number().int().min(1),
    maxScore: z.number().int().min(1),
    evaluationCriteria: z.object({
      errors: z.object({
        weight: z.number().positive(),
        description: z.string()
      }),
      doubts: z.object({
        weight: z.number().positive(),
        description: z.string()
      }),
      tajweed: z.object({
        weight: z.number().positive(),
        description: z.string()
      })
    })
  })),
  totalScore: z.number().int().default(100),
  passingScore: z.number().int().default(80)
});

// ========== إنشاء النموذج ==========

export async function createExamModel(input: z.infer<typeof createExamModelInput>) {
  // 1. التحقق من المستخدم والصلاحية
  const user = await getCurrentUser();
  if (!user) throw new Error("غير مسجل دخول");
  
  if (user.role !== Role.TEST_SPECIALIST && user.role !== Role.ADMIN) {
    throw new Error("ليس لديك صلاحية إنشاء نماذج");
  }
  
  // 2. التحقق من البيانات
  const validated = createExamModelInput.parse(input);
  
  // 3. التحقق من عدم التكرار (رقم النموذج فريد)
  const existing = await prisma.examModel.findUnique({
    where: { 
      institutionId_number: {
        institutionId: user.institutionId || undefined,
        number: validated.number
      }
    }
  });
  
  if (existing) {
    throw new Error(`النموذج رقم ${validated.number} موجود بالفعل`);
  }
  
  // 4. إنشاء النموذج
  const model = await prisma.examModel.create({
    data: {
      number: validated.number,
      startSurah: validated.startSurah,
      startVerse: validated.startVerse,
      startText: validated.startText,
      endSurah: validated.endSurah,
      endVerse: validated.endVerse,
      endText: validated.endText,
      sections: validated.sections,
      scoringRules: {
        errors: 1.0,
        doubts: 0.5,
        tajweed: 0.25
      },
      totalScore: validated.totalScore,
      passingScore: validated.passingScore,
      institutionId: user.institutionId
    }
  });
  
  // 5. تسجيل في Audit Log
  await createAuditLog({
    action: "CREATE_EXAM_MODEL",
    resourceType: "ExamModel",
    resourceId: model.id,
    userId: user.id,
    changes: { modelNumber: model.number }
  });
  
  // 6. الإشعار
  await createNotification({
    userId: user.id,
    type: "MODEL_CREATED",
    title: `تم إنشاء النموذج رقم ${model.number}`,
    message: `من ${model.startSurah}:${model.startVerse} إلى ${model.endSurah}:${model.endVerse}`,
    relatedId: model.id
  });
  
  return { success: true, modelId: model.id };
}

// ========== تحديث النموذج ==========

export async function updateExamModel(
  modelId: string, 
  input: Partial<z.infer<typeof createExamModelInput>>
) {
  const user = await getCurrentUser();
  if (!user) throw new Error("غير مسجل دخول");
  
  // التحقق من الملكية
  const model = await prisma.examModel.findUniqueOrThrow({
    where: { id: modelId }
  });
  
  if (model.institutionId !== user.institutionId && user.role !== Role.ADMIN) {
    throw new Error("لا يمكنك تعديل نموذج جهة أخرى");
  }
  
  // تحديث البيانات المسموحة فقط
  const updateData: any = {};
  if (input.startSurah) updateData.startSurah = input.startSurah;
  if (input.startVerse) updateData.startVerse = input.startVerse;
  if (input.startText !== undefined) updateData.startText = input.startText;
  if (input.endSurah) updateData.endSurah = input.endSurah;
  if (input.endVerse) updateData.endVerse = input.endVerse;
  if (input.endText !== undefined) updateData.endText = input.endText;
  if (input.sections) updateData.sections = input.sections;
  if (input.totalScore) updateData.totalScore = input.totalScore;
  if (input.passingScore) updateData.passingScore = input.passingScore;
  
  const updated = await prisma.examModel.update({
    where: { id: modelId },
    data: updateData
  });
  
  // Audit Log + Notification
  await createAuditLog({
    action: "UPDATE_EXAM_MODEL",
    resourceType: "ExamModel",
    resourceId: modelId,
    userId: user.id,
    changes: updateData
  });
  
  return { success: true };
}

// ========== الحصول على نموذج ==========

export async function getExamModel(modelId: string) {
  const model = await prisma.examModel.findUniqueOrThrow({
    where: { id: modelId }
  });
  
  return {
    ...model,
    displayName: `${model.startSurah}:${model.startVerse} → ${model.endSurah}:${model.endVerse}`
  };
}

// ========== قائمة النماذج ==========

export async function getExamModels(filters?: {
  institutionId?: string;
  search?: string;
  page?: number;
}) {
  const where: any = {};
  
  if (filters?.institutionId) {
    where.institutionId = filters.institutionId;
  }
  
  if (filters?.search) {
    where.OR = [
      { startSurah: { contains: filters.search, mode: 'insensitive' } },
      { endSurah: { contains: filters.search, mode: 'insensitive' } }
    ];
  }
  
  const page = filters?.page || 1;
  const take = 20;
  const skip = (page - 1) * take;
  
  const [models, total] = await Promise.all([
    prisma.examModel.findMany({
      where,
      skip,
      take,
      orderBy: { number: 'asc' }
    }),
    prisma.examModel.count({ where })
  ]);
  
  return {
    models: models.map(m => ({
      ...m,
      displayName: `${m.startSurah}:${m.startVerse} → ${m.endSurah}:${m.endVerse}`
    })),
    pagination: { page, total, pages: Math.ceil(total / take) }
  };
}

### ✅ المهمة 8: الإشعارات المتقدمة

**الملف:** `lib/services/notification-service.ts` (جديد)

```typescript
import { prisma } from '@/lib/db';
import { Resend } from 'resend';
import { Twilio } from 'twilio';

const resend = new Resend(process.env.RESEND_API_KEY);
const twilio = new Twilio(process.env.TWILIO_ACCOUNT_SID, process.env.TWILIO_AUTH_TOKEN);

export const notificationService = {
  // إشعار بريد إلكتروني
  async sendEmail(to: string, subject: string, html: string) {
    try {
      return await resend.emails.send({
        from: 'notifications@exam-platform.com',
        to,
        subject,
        html
      });
    } catch (error) {
      console.error('Failed to send email:', error);
      throw error;
    }
  },

  // رسالة نصية
  async sendSMS(to: string, body: string) {
    try {
      return await twilio.messages.create({
        from: process.env.TWILIO_PHONE,
        to,
        body
      });
    } catch (error) {
      console.error('Failed to send SMS:', error);
      throw error;
    }
  },

  // إشعار متعدد القنوات
  async notifyMultiChannel(
    userId: string,
    type: string,
    title: string,
    message: string
  ) {
    const user = await prisma.user.findUniqueOrThrow({ where: { id: userId } });

    // 1. إشعار في-المنصة
    await prisma.notification.create({
      data: {
        userId,
        type,
        title,
        message
      }
    });

    // 2. بريد إلكتروني
    if (user.email) {
      await this.sendEmail(
        user.email,
        title,
        `<p>${message}</p><p><a href="http://exam-platform.com/dashboard">فتح المنصة</a></p>`
      );
    }

    // 3. رسالة نصية (اختياري)
    if (user.phone && process.env.TWILIO_ENABLED === 'true') {
      await this.sendSMS(user.phone, `${title}\n${message}`);
    }
  }
};
```
